Load & Query the Registry

model_cards.py
stdlib only 
"""Load and query the geospatial AI model-card registry."""
import json
from dataclasses import dataclass
from typing import List

HIGH_RISK_TERMS = ("high", "military", "surveillance", "misinformation", "targeting")

@dataclass(frozen=True)
class ModelCard:
    id: str
    algorithm: str
    intended_use: str
    business_case: str
    misuse_risk: str
    compliance: str

    @classmethod
    def from_dict(cls, row: dict) -> "ModelCard":
        return cls(
            id=row["Id"],
            algorithm=row["AI Algorithm"],
            intended_use=row["Intended Use"],
            business_case=row["Business Case"],
            misuse_risk=row["Misuse Risk"],
            compliance=row["Compliance Alignment"],
        )

    def is_high_risk(self) -> bool:
        blob = f"{self.misuse_risk} {self.compliance}".lower()
        return any(term in blob for term in HIGH_RISK_TERMS)

def load(path: str = "model_cards.json") -> List[ModelCard]:
    with open(path, encoding="utf-8") as fh:
        return [ModelCard.from_dict(row) for row in json.load(fh)]

if __name__ == "__main__":
    cards = load()
    flagged = [c for c in cards if c.is_high_risk()]
    print(f"{len(cards)} cards loaded, {len(flagged)} flagged for ethical review")
    for card in flagged[:10]:
        print(f"  {card.id:<12} {card.algorithm:<28} {card.misuse_risk}")

Jackson · Java 17+ 
package geo.ai.modelcards;

import com.fasterxml.jackson.annotation.JsonProperty;
import com.fasterxml.jackson.core.type.TypeReference;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.io.File;
import java.util.List;
import java.util.stream.Collectors;

public record ModelCard(
    @JsonProperty("Id") String id,
    @JsonProperty("AI Algorithm") String algorithm,
    @JsonProperty("Intended Use") String intendedUse,
    @JsonProperty("Business Case") String businessCase,
    @JsonProperty("Misuse Risk") String misuseRisk,
    @JsonProperty("Compliance Alignment") String compliance
) {
    private static final List<String> HIGH_RISK =
        List.of("high", "military", "surveillance", "misinformation", "targeting");

    public boolean isHighRisk() {
        String blob = (misuseRisk + " " + compliance).toLowerCase();
        return HIGH_RISK.stream().anyMatch(blob::contains);
    }

    public static void main(String[] args) throws Exception {
        ObjectMapper mapper = new ObjectMapper();
        List<ModelCard> cards = mapper.readValue(
            new File("model_cards.json"),
            new TypeReference<List<ModelCard>>() {});
        
        System.out.printf("%d model cards loaded%n", cards.size());
        cards.stream().filter(ModelCard::isHighRisk).limit(10)
             .forEach(c -> System.out.printf("  %-12s %-28s %s%n",
                     c.id(), c.algorithm(), c.misuseRisk()));
    }
}
main.go
stdlib only 
package main

import (
    "encoding/json"
    "fmt"
    "os"
    "strings"
)

type ModelCard struct {
    ID           string `json:"Id"`
    Algorithm    string `json:"AI Algorithm"`
    IntendedUse  string `json:"Intended Use"`
    BusinessCase string `json:"Business Case"`
    MisuseRisk   string `json:"Misuse Risk"`
    Compliance   string `json:"Compliance Alignment"`
}

var highRiskTerms = []string{"high", "military", "surveillance", "misinformation", "targeting"}

func (m ModelCard) IsHighRisk() bool {
    blob := strings.ToLower(m.MisuseRisk + " " + m.Compliance)
    for _, term := range highRiskTerms {
        if strings.Contains(blob, term) {
            return true
        }
    }
    return false
}

func main() {
    raw, _ := os.ReadFile("model_cards.json")
    var cards []ModelCard
    json.Unmarshal(raw, &cards)
    
    flagged := 0
    for _, c := range cards {
        if c.IsHighRisk() {
            flagged++
            fmt.Printf("  %-12s %-28s %s\n", c.ID, c.Algorithm, c.MisuseRisk)
        }
    }
    fmt.Printf("%d cards loaded, %d flagged for ethical review\n", len(cards), flagged)
}

use serde::{Deserialize, Serialize};
use std::fs::File;
use std::io::{self, BufReader};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ModelCard {
    #[serde(rename = "Id")]
    pub id: String,
    #[serde(rename = "AI Algorithm")]
    pub algorithm: String,
    #[serde(rename = "Misuse Risk")]
    pub misuse_risk: String,
    #[serde(rename = "Compliance Alignment")]
    pub compliance: String,
}

const HIGH_RISK_TERMS: [&str; 5] = [
    "high", "military", "surveillance", "misinformation", "targeting",
];

impl ModelCard {
    pub fn is_high_risk(&self) -> bool {
        let blob = format!("{} {}", self.misuse_risk, self.compliance).to_lowercase();
        HIGH_RISK_TERMS.iter().any(|term| blob.contains(term))
    }
}

pub fn load(path: &str) -> io::Result<Vec<ModelCard>> {
    let file = File::open(path)?;
    let reader = BufReader::new(file);
    serde_json::from_reader(reader)
        .map_err(|e| io::Error::new(io::ErrorKind::InvalidData, e))
}

fn main() -> io::Result<()> {
    let cards = load("model_cards.json")?;
    let flagged: Vec<&ModelCard> = cards.iter()
        .filter(|c| c.is_high_risk()).collect();
    
    println!("{} cards loaded, {} flagged for review", cards.len(), flagged.len());
    Ok(())
}

circe · Scala 3 
import io.circe.Decoder
import io.circe.parser.decode
import scala.io.Source
import scala.util.Using

final case class ModelCard(
    id: String,
    algorithm: String,
    intendedUse: String,
    businessCase: String,
    misuseRisk: String,
    compliance: String
)

object ModelCard {
  private val HighRiskTerms = 
    List("high", "military", "surveillance", "misinformation", "targeting")

  implicit val decoder: Decoder[ModelCard] = Decoder.instance { c =>
    for {
      id           <- c.get[String]("Id")
      algorithm    <- c.get[String]("AI Algorithm")
      misuseRisk   <- c.get[String]("Misuse Risk")
      compliance   <- c.get[String]("Compliance Alignment")
    } yield ModelCard(id, algorithm, "", "", misuseRisk, compliance)
  }

  def isHighRisk(card: ModelCard): Boolean = {
    val blob = s"${card.misuseRisk} ${card.compliance}".toLowerCase
    HighRiskTerms.exists(blob.contains)
  }
}

object Registry extends App {
  ModelCard.load("model_cards.json") match {
    case Right(cards) =>
      val flagged = cards.filter(ModelCard.isHighRisk)
      println(s"${cards.size} cards loaded, ${flagged.size} flagged")
    case Left(err) => System.err.println(s"Error: $err")
  }
}
